--多条件判断
DECLARE
  V_A NUMBER := &输入数字;
BEGIN
  IF V_A = 1 AND 1 = 2 THEN
    --条件判断1
    --执行逻辑1
    DBMS_OUTPUT.put_line('上班');
  ELSIF V_A = 2 THEN
    --上面条件不成立时,才会执行下一个条件判断
    --条件判断2
    --执行逻辑2
    DBMS_OUTPUT.put_line('下班');
  ELSIF V_A = 3 THEN
    --判断条件3
    --执行逻辑3
    DBMS_OUTPUT.put_line('加班');
  ELSE
    --上述所有条件都不成立时执行ELSE
    --执行逻辑4
    DBMS_OUTPUT.put_line('放假');
  END IF;
END;

传入一个员工编号
操作EMP1表
如果这个员工的岗位是ANALYST或者CLERK或者SALESMAN，
就给该员工加500元奖金
如果这个员工的岗位是MANAGER，就给该员工加800块奖金
如果这个员工的岗位是PRESIDENT，不做处理

DECLARE
  V_EMPNO NUMBER := &员工编号; --7788
  V_JOB   VARCHAR2(20);
BEGIN
  SELECT JOB INTO V_JOB FROM EMP WHERE EMPNO = V_EMPNO;
  IF V_JOB IN ('ANALYST', 'CLERK', 'SALESMAN') THEN
    UPDATE EMP1 SET COMM = NVL(COMM, 0) + 500 WHERE EMPNO = V_EMPNO;
  ELSIF V_JOB = 'MANAGER' THEN
    UPDATE EMP1 SET COMM = NVL(COMM, 0) + 800 WHERE EMPNO = V_EMPNO;
  END IF;
END;

小练习：
传入一个员工编号
如果这个员工工资小于1000，就给他加300奖金
如果这个员工工资1000到2000，就给他加200奖金
如果这个员工工资大于2000，就给他加100奖金
输出员工编号，员工名字，员工工资，员工奖金(涨薪后)


DECLARE
  V_A EMP%ROWTYPE;
BEGIN
  V_A.EMPNO := &员工编号; --7788
  SELECT * INTO V_A FROM EMP1 WHERE EMPNO = V_A.EMPNO;
  IF V_A.SAL < 1000 THEN
    V_A.COMM := NVL(V_A.COMM, 0) + 300;
  ELSIF V_A.SAL <= 2000 THEN
    V_A.COMM := NVL(V_A.COMM, 0) + 200;
  ELSE
    V_A.COMM := NVL(V_A.COMM, 0) + 100;
  END IF;
  UPDATE EMP1 SET COMM = V_A.COMM WHERE EMPNO = V_A.EMPNO;
  DBMS_OUTPUT.put_line(V_A.EMPNO || V_A.ENAME || V_A.SAL || '-' ||
                       V_A.COMM);
END;

--IF嵌套
传入一个员工编号，如果该员工是'ANALYST','CLERK','SALESMAN'
岗位就加500工资，如果该员工工资在3000以上则再加500，
如果是MANAGER岗位则加800工资

DECLARE
  V_EMPNO NUMBER := &输入编号;
  V_JOB   VARCHAR2(20);
  V_SAL   NUMBER;
BEGIN
  SELECT JOB, SAL INTO V_JOB, V_SAL FROM EMP WHERE EMPNO = V_EMPNO;
  IF V_JOB IN ('ANALYST', 'CLERK', 'SALESMAN') THEN
    --7788 CLERK 2501
    IF V_SAL > 3000 THEN
      --外层条件成立时,才会执行嵌套的IF判断
      V_SAL := V_SAL + 1000;
    ELSE
      V_SAL := V_SAL + 500;
    END IF;
    --V_SAL := V_SAL + 500;--岗位的涨薪
  ELSIF V_JOB = 'MANAGER' THEN
    V_SAL := V_SAL + 800;
  END IF;
  DBMS_OUTPUT.put_line(V_EMPNO || V_JOB || V_SAL);
END;

输入两个员工编号将工资高的这个员工的名字打印出来。
(相同打印两人姓名  禁止输入两个相同员工编号)

DECLARE
  V_A EMP%ROWTYPE;
  V_B EMP%ROWTYPE;
BEGIN
  V_A.EMPNO := &编号1;
  V_B.EMPNO := &编号2;
  IF V_A.EMPNO = V_B.EMPNO THEN
    DBMS_OUTPUT.put_line('禁止输入两个相同员工编号');
  ELSE
    SELECT * INTO V_A FROM EMP WHERE EMPNO = V_A.EMPNO;
    SELECT * INTO V_B FROM EMP WHERE EMPNO = V_B.EMPNO;
    IF V_A.SAL > V_B.SAL THEN
      DBMS_OUTPUT.put_line(V_A.ENAME);
    ELSIF V_B.SAL > V_A.SAL THEN
      DBMS_OUTPUT.put_line(V_B.ENAME);
    ELSE
      DBMS_OUTPUT.put_line(V_A.ENAME || V_B.ENAME);
    END IF;
  END IF;
END;

DECLARE

  --开始循环
  LOOP
    --设置退出循环的条件

        V_A := V_A + 1;
    -- END IF;

  END LOOP; --结束循环
END;

  V_A NUMBER := 0;

  END LOOP;
END;

2.假设现在有鸡和兔子，一共有35只，它们的脚一共有94个，
使用一个for循环计算出鸡和兔子分别有多少只。
穷举法

BEGIN
  --假设全是鸡
  FOR I IN 0 .. 35 LOOP
    IF I * 2 + (35 - I) * 4 = 94 THEN
      DBMS_OUTPUT.put_line('鸡: ' || I || ' 兔: ' || (35 - I));
    END IF;
  END LOOP;
END;

4.从今天开始存钱，今天存1分，明天2分，后天4分，每天翻倍，

日存款  总存款  天

DECLARE
  V_A NUMBER := 0; --总存款
  V_B NUMBER := 1; --日存款
  V_C NUMBER := 0; --天
BEGIN
  WHILE V_A < 100000000 LOOP
    V_A := V_A + V_B; --累加日存款
    V_C := V_C + 1; --天数
    DBMS_OUTPUT.put_line('天: ' || V_C || ' 日存 ' || V_B || ' 总 ' || V_A);
    V_B := V_B * 2; --日存款翻倍
  END LOOP;
  DBMS_OUTPUT.put_line('天: ' || V_C);
END;

--外层循环每循环一次,嵌套的循环会循环至上限
DECLARE
  V_A NUMBER := 0; --星号
  V_B NUMBER := 0;
  V_C NUMBER := 0; --空格
  V_D NUMBER := &输入行号;
BEGIN
  WHILE V_B < V_D LOOP
    --控制行
    --------控制空格-------------
    WHILE V_C  < V_B LOOP
      DBMS_OUTPUT.put(' ');
      V_C := V_C + 1;
    END LOOP;
    ------控制星号的循环--------
    WHILE V_A < 2 * (V_D - V_B) - 1 LOOP
      --控制列
      V_A := V_A + 1;
      DBMS_OUTPUT.put('*');
    END LOOP;
    DBMS_OUTPUT.new_line;
    ----------------------
    V_A := 0;
    V_C := 0;
    V_B := V_B + 1;
  END LOOP;
END;
