SELECT *
  FROM (SELECT A.PAV, A.BRN, SUM(A.A_NUM) A_NUMS, SUM(A.B_NUM) B_NUMS
          FROM A   
         GROUP BY A.PAV, A.BRN) AA
  LEFT JOIN B
    ON AA.PAV = B.PAV
   AND AA.BRN = B.BRN
  LEFT JOIN (SELECT C.IP_WORD, SUM(C.C_NUM) C_NUMS FROM C GROUP BY C.IP_WORD) CC
    ON CC.IP_WORD = B.IP_WORD;

--1、按name分组取val最大的值所在行的数据 写出至少五种方法


--1.0
SELECT *
  FROM (SELECT T.*, MAX(T.VAL) OVER(PARTITION BY T.NAME) ZG
          FROM t_groupBy_orderby_limitn T)
 WHERE VAL = ZG;

--2.0
SELECT *
  FROM (SELECT ROW_NUMBER() OVER(PARTITION BY T.NAME ORDER BY T.VAL DESC) PM,
               T.*
          FROM t_groupBy_orderby_limitn T)
 WHERE PM = 1;
 
--3.0
SELECT *
  FROM t_groupBy_orderby_limitn T
 WHERE (T.NAME, T.VAL) IN (SELECT A.NAME, MAX(A.VAL)
                             FROM t_groupBy_orderby_limitn A
                            GROUP BY A.NAME);

--4.0
SELECT *
  FROM t_groupBy_orderby_limitn A
  LEFT JOIN (SELECT B.NAME, MAX(B.VAL) ZD
               FROM t_groupBy_orderby_limitn B
              GROUP BY B.NAME) BB
    ON A.NAME = BB.NAME
   AND A.VAL = BB.ZD
 WHERE BB.NAME IS NOT NULL;

--5.0

SELECT *
  FROM t_groupBy_orderby_limitn A
 WHERE 1 > ( --求B表比A表中VAL大的数量
            SELECT COUNT(1) /*0*/
              FROM t_groupBy_orderby_limitn B
             WHERE A.NAME = B.NAME --伪分组
               AND B.VAL /*532*/

            );
