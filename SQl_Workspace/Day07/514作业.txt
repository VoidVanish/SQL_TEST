求累加到当前行的金额是一百 使用sum（）over开窗累加 

求得所有金额的累加 我要得到的是金额刚好超过一百的那行数据 在此条件下 求得的累加金额表可以进行过滤判断
找到累加金额刚好超过一百的那条 单独输出一行6 1 101 的数据 值判断无效


SELECT ROW_NUMBER() OVER(ORDER BY T1.ID) PM


SELECT T2.*
  FROM (SELECT T1.*, ROW_NUMBER() OVER(ORDER BY T1.ID) PM
          FROM (SELECT ID,
                       金额,
                       SUM(金额) OVER(ORDER BY ID ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) 累加值
                  FROM G0328) T1
         WHERE T1.累加值 > 100) T2
 WHERE T2.PM = 1

 逻辑分析 需要新建两个字段 分别是银行存款、存入总额、取现总额 
                                     存入总额 和取现在流水表当中
                      
               
SELECT ACCT_ID,TRADE_TYPE,SUM(AMT) 存取总额
FROM AMT_INF
GROUP BY ACCT_ID,TRADE_TYPE

列转行 把存入、取现作为字段 总额作为数值

1 存取总额临表
SELECT *
FROM (SELECT ACCT_ID,TRADE_TYPE,SUM(AMT) 存取总额
     FROM AMT_INF
     GROUP BY ACCT_ID,TRADE_TYPE
) T1 PIVOT (
  SUM(T1.存取总额) FOR TRADE_TYPE IN('存入' AS 存入总金额 ,'取现' AS 取现总金额)
  );

BAL表需要聚合各个人的账户总金额
2 账户总存款临表

SELECT ACCT_ID,CUST_ID,SUM(BAL)
FROM bal_info
GROUP BY ACCT_ID,CUST_ID

3关联所有表
SELECT CT.CUST_ID,CT.CUST_NAME,SUM(CU.BAL) BAL,SUM(AC.存入总金额) 存入总金额,SUM(AC.取现总金额) 取现总金额
  FROM cust_tal CT
  LEFT JOIN (SELECT ACCT_ID, CUST_ID, SUM(BAL) BAL
               FROM bal_info
              GROUP BY ACCT_ID, CUST_ID) CU
    ON CU.CUST_ID = CT.CUST_ID
  LEFT JOIN (SELECT *
               FROM (SELECT ACCT_ID, TRADE_TYPE, SUM(AMT) 存取总额
                       FROM AMT_INF
                      GROUP BY ACCT_ID, TRADE_TYPE) T1
             PIVOT(SUM(T1.存取总额)
                FOR TRADE_TYPE IN('存入' AS 存入总金额,
                                 '取现' AS 取现总金额))) AC
    ON CU.ACCT_ID = AC.ACCT_ID
    GROUP BY CT.CUST_ID,CUST_NAME
    ORDER BY CUST_ID
