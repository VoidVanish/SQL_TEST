-- 4.3、查询不及格的课程并按课程号从大到小排列

SELECT * FROM SCORE WHERE  SSCORE < 60 ORDER BY SCOURE DESC;

-- 4.4、查询每门课程的平均成绩，结果按平均成绩升序排序，
--平均成绩相同时，按课程号降序排列

SELECT SCOURE, AVG(SSCORE) PJ
  FROM SCORE
 GROUP BY SCOURE
 ORDER BY PJ ASC, SCOURE DESC;

-- 4.5、检索课程编号为“0004”且分数小于60的学生学号，
--结果按按分数降序排列

SELECT *
  FROM SCORE
 WHERE SCOURE = 4
   AND SSCORE < 60
 ORDER BY SSCORE DESC;

--行列转换

--列转行
Y      Q1   Q2    Q3    Q4
2015  100   110   140   100
2016  200   150   100   300

SELECT * FROM TEST_TABLE;

--CASE WHEN
SELECT Y,
SUM(CASE WHEN Q=1 THEN AMT END) AS Q1,
SUM(CASE WHEN Q=2 THEN AMT END) AS Q2,
SUM(CASE WHEN Q=3 THEN AMT END) AS Q3,
SUM(CASE WHEN Q=4 THEN AMT END) AS Q4
FROM TEST_TABLE GROUP BY Y;

--DECODE(字段名,判断条件,成立返回,不成立返回值(默认空))
--        CASE   WHEN     THEN       ELSE 
--只能进行等值判断
SELECT DECODE(ENAME,'SCOTT','张三'),ENAME
FROM EMP;

--判断多个值
SELECT DECODE(ENAME, 'SCOTT', '张三', 'SMITH', '李四', 'KING', '王五'),
       ENAME
  FROM EMP;

--DECODE列转行

SELECT Y,
       MAX(DECODE(Q, 1, AMT)) Q1,
       MAX(DECODE(Q, 2, AMT)) Q2,
       MAX(DECODE(Q, 3, AMT)) Q3,
       MAX(DECODE(Q, 4, AMT)) Q4
  FROM TEST_TABLE
 GROUP BY Y;

--PIVOT列转行函数
/*
TABLE_NAME PIVOT(
聚合函数(数据字段) FOR 关键字段 IN(关键字段的值)
)*/
SELECT * FROM TEST_TABLE 
PIVOT(SUM(AMT) FOR Q IN(1 AS Q1,2 AS Q2,3 AS Q3,4 AS Q4));
