--开窗函数(最重要)OVER(窗口说明)

--聚合类开窗函数SUM()OVER() AVG COUNT MAX MIN

SELECT SUM(SAL) OVER(), MAX(SAL) OVER(), DEPTNO FROM EMP;

--PARTITION BY 分组不去重
SELECT SUM(SAL)OVER(PARTITION BY DEPTNO),
MIN(SAL)OVER(PARTITION BY DEPTNO),
SAL,DEPTNO FROM EMP;

--ORDER BY 排序 SUM+ORDER BY等于累加 ******
--第一行-最后一行 => 第一行-当前行
SELECT SUM(SAL) OVER(ORDER BY EMPNO),
       MAX(SAL) OVER(ORDER BY EMPNO) 截止到当前行的最大值,
       SAL,
       EMPNO
  FROM EMP;

CURRENT ROW:当前行
n PRECEDING:上几行
n FOLLOWING:下几行
UNBOUNDED PRECEDING:第一行
UNBOUNDED FOLLOWING:最后一行

--控制聚合范围 ROWS BETWEEN AND
SELECT SUM(SAL) OVER(ORDER BY EMPNO ROWS BETWEEN 1 PRECEDING AND 1 FOLLOWING) S1,
       SUM(SAL) OVER(ORDER BY EMPNO ROWS BETWEEN CURRENT ROW AND UNBOUNDED FOLLOWING) S2,
       SUM(SAL) OVER(ORDER BY EMPNO ROWS BETWEEN 1 PRECEDING AND current row) S3,
       SAL,
       EMPNO
  FROM EMP;

--分组累加
SELECT SUM(SAL) OVER(PARTITION BY DEPTNO ORDER BY EMPNO),
       SAL,
       EMPNO,
       DEPTNO
  FROM EMP;


--连续登录问题
有登陆表logintest, 表中字段：user_id 用户名,log_date 登录日期
查找连续登录三天的用户信息
--聚合函数不能嵌套开窗函数,开窗函数可以嵌套聚合函数
SELECT COUNT(JG), USER_ID
  FROM (SELECT L.USER_ID,
               L.LOG_DATE,
               ROW_NUMBER() OVER(PARTITION BY L.USER_ID ORDER BY L.LOG_DATE) PM,
               L.LOG_DATE - ROW_NUMBER() OVER(PARTITION BY L.USER_ID ORDER BY L.LOG_DATE) JG
          FROM LOGINTEST L)
 GROUP BY USER_ID, JG
HAVING COUNT(JG) > 2;


--排名类开窗函数
--函数名()OVER(必须有ORDER BY )
--排名函数的区别*******
RANK 可能生成一组重复且不连续的序号 123356
DENSE_RANK 可能生成一组重复且连续的序号,序号总数会变少 123345
ROW_NUMBER 生成一组连续且不重复的序号,重复值会给不同的序号 123456

SELECT ROW_NUMBER() OVER(ORDER BY SAL),
       RANK() OVER(ORDER BY SAL),
       DENSE_RANK() OVER(ORDER BY SAL),
       SAL
  FROM EMP;

--组内排序
SELECT ROW_NUMBER() OVER(PARTITION BY DEPTNO ORDER BY SAL), SAL, DEPTNO
  FROM EMP;

--2.泛用性  开窗函数不能过滤
SELECT *
  FROM (SELECT E.*,
               --子查询中使用函数一定要起别名
               ROW_NUMBER() OVER(ORDER BY SAL DESC) PM
          FROM EMP E) Q
