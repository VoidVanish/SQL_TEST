题3（15分）现有一个数据表MUSEUM，记录了一个小型博物馆每天的访客数量： 10
RECORD_DATE	VISITORS
20160705	371
20160706	128
20160709	85
20160710	246
20160712	169

博物馆会不定期闭馆，闭馆当天不会向该表添加记录，例如上面缺少的20160707、20160708、20160711三天为闭馆。尝试以查询SQL或存储过程的形式，求该博物馆曾最长连续闭馆多少天。（时间范围以RECORD_DATE字段的最小值、最大值为准）
以上面5行数据为例，求出的结果为2，即20160705～20160712这个时间范围内，最长闭馆2天，对应20160707、20160708；如果表中的日期数据是连续的，求出的结果为0。

--求曾经连续闭馆最长时间 题目表示最长闭馆时间是两天 最短为一天 每个开放时间的间隔
--就是闭馆时间 用循环 判断时间与日期计数器是否一致 若为连续开放 则不是闭馆 闭馆计数器不加1 
SELECT * FROM MUSEUM

  --存储过程
  CREATE OR REPLACE PROCEDURE P_MUSEUM IS
  CURSOR C_A IS
    SELECT RECORD_DATE, VISITORS FROM MUSEUM;
  --定义闭馆计数器
  V_BG NUMBER:=0;
  --定义日期核对器
  --判断musum表与当前核对器的日期是否相同 DATE;
  V_A    MUSEUM%ROWTYPE; --第一行
  V_DATE DATE; ---核对器
BEGIN
  --获取开始日期 
  OPEN C_A;
  FETCH C_A
    INTO V_A;
  V_DATE := V_A.RECORD_DATE;
  CLOSE C_A;
  --DBMS_OUTPUT.put_line(V_DATE);
  DBMS_OUTPUT.put_line(V_BG);
  --判断musum表与当前核对器的日期是否相同
  FOR I IN C_A LOOP
    --第一个数已存在
  
    IF V_DATE = I.RECORD_DATE THEN
      V_DATE := I.RECORD_DATE; --相同则当前日期等于计数器
    ELSIF V_BG < (I.RECORD_DATE) -( V_DATE) - 1 THEN
      V_BG   := I.RECORD_DATE - V_DATE - 1;
      V_DATE := I.RECORD_DATE + 1; -- 不相同则赋值给闭馆计数器
    ELSE
      V_DATE := I.RECORD_DATE;
    END IF;
    -- END IF;
  END LOOP;
  DBMS_OUTPUT.put_line('最长闭馆天数为  '||  V_BG ||'  天');
END;

--调用
BEGIN 
  P_MUSEUM();
END;




题5(10分)已知一个客户手机号码表，T_CUSTMOBILE(CUSTNO NUMBER,MOBILENO VARCHAR2(20));请通过程序实现对该表数据进行逐条遍历，找出手机号码合法的客户号，并将客户号保存到T_TMPCUST(CUSTNO NUMBER)表中，并考虑当数据量较大的情况下实现分批提交事务。 8
已知手机号码合法的检查规则：号码由长度11位的数字组成，并由13、15、18开头，请优先考虑使用正则表达式。
--筛选对的手机号
DROP TABLE T_TMPCUST
CREATE TABLE T_TMPCUST(CUSTNO NUMBER);
SELECT * FROM T_TMPCUST
--总
CREATE TABLE T_TMPCUST AS
SELECT T1.CUSTNO
          FROM (SELECT T.*, ROWNUM RN
                  FROM (SELECT CUSTNO, MOBILENO
                          FROM T_CUSTMOBILE
                         WHERE REGEXP_LIKE(T_CUSTMOBILE.CUSTNO,
                                           '^1[358][0-9]{9}$')) T) T1
         WHERE T1.RN <= 50;

--展示每个季度

--根据季度分组

SELECT T1.AGENCYNO,T1.SALECOUNT,T1.季度,SUM(T1.SALECOUNT)OVER(PARTITION BY T1.季度) 销售数量
