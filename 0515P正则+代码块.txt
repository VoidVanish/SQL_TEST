  FROM (SELECT G.ID,
               G.金额,
               (SELECT SUM(B.金额) FROM G0328 B WHERE B.ID <= G.ID) 累加值
          FROM G0328 G) A

--聚合函数只能和分组后的字段或者其他聚合函数一起使用

SELECT C.CUST_ID,
       C.CUST_NAME,
       SUM(BAL) BAL,
       SUM(AA.存入总金额) 存入总金额,
       SUM(AA.取现总金额) 取现总金额
  FROM CUST_TAL C
  LEFT JOIN BAL_INFO B
    ON C.CUST_ID = B.CUST_ID
  LEFT JOIN (SELECT A.ACCT_ID,
                    SUM(CASE
                          WHEN A.TRADE_TYPE = '存入' THEN
                           AMT
                        END) 存入总金额,
                    SUM(CASE
                          WHEN A.TRADE_TYPE = '取现' THEN
                           AMT
                        END) 取现总金额
               FROM AMT_INF A
              GROUP BY A.ACCT_ID) AA
    ON AA.ACCT_ID = B.ACCT_ID
 GROUP BY C.CUST_ID, C.CUST_NAME;

+ 匹配一次或者无数次

? 匹配0次或者1次

|  表示或者的意思
() 表示一组正则表达式

--PL/SQL 存储过程
  oracle PL/SQL语言(Procedural Language/SQL)是结合了结构化查询与oracle自身
  过程控制为一体的强大语言。
  语法结构：
    DECLARE
      --declaration statements (1)
    BEGIN
      --executable statements (2)
    EXCEPTION
      --exception statements (3)
    END;
  语法解析：
    (1)声明部分：声明部分包含了变量和常量的定义，这部分由关键字DECLARE开始，
      如果不声明变量或者常量，可以省略这部分。
    (2)执行部分：执行部分是PL/SQL块的指令部分，由关键字BEGIN开始，关键字END结束。
      所有的可执行PL/SQL语句都放在这一部分，该部分执行命令并操作变量。
      其他的PL/SQL块可以作为子块嵌套在该部分。PL/SQL块的执行部分是必选的。
      注意END关键字后面用分号结尾。
    (3)异常处理部分：该部分是可选的，该部分用EXCEPTION关键字把可执行部分分成两个小部分，
      之前的程序是正常运行的程序，一旦出现异常就跳转到异常部分执行。

DECLARE

  --声明变量的初始赋值 
  V_A VARCHAR2(20) := '小明';
  --外部向变量赋值 &提示符

  DBMS_OUTPUT.put_line('年龄: ' || V_C || ' 生日: ' ||
                       TO_CHAR(V_D, 'YYYYMMDD'));

--传入一个半径,求出圆的面积和周长

DECLARE
  V_R NUMBER := &半径;
  C_A CONSTANT NUMBER := 3.14;
  V_A NUMBER;
BEGIN
  --面积
  V_A := POWER(V_R, 2) * C_A;
  DBMS_OUTPUT.put_line('面积: ' || V_A);
  --周长
  V_A := 2 * C_A * V_R;
  DBMS_OUTPUT.put_line('周长: ' || V_A);
END;

--异常处理部分
--系统的预定义异常：
--数据库给某个错误赋予了错误的编号和错误的名字(21种)
no_data_found   查询没有数据
too_many_rows  查询数据太多
value_error  赋值的错误
zero_divide  除数为0
......

DECLARE
  V_EMPNO NUMBER := &编号;
  V_ENAME VARCHAR2(20);
  V_SAL   NUMBER;
BEGIN
  SELECT ENAME INTO V_SAL FROM EMP WHERE EMPNO = V_EMPNO;
  DBMS_OUTPUT.put_line(V_ENAME);
EXCEPTION
  --WHEN 异常名 THEN
  WHEN NO_DATA_FOUND THEN
    --SQLCODE 报错编号 SQLERRM报错信息
    DBMS_OUTPUT.put_line(SQLERRM);
  WHEN VALUE_ERROR THEN
    DBMS_OUTPUT.put_line(SQLERRM);
  WHEN TOO_MANY_ROWS THEN
    DBMS_OUTPUT.put_line(SQLERRM);
END;

--OTHERS 包含所有异常情况
--编译时期报错  运行时期报错(异常)

  WHEN OTHERS THEN
    DBMS_OUTPUT.put_line(SQLERRM);
