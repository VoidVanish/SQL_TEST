---存储过程
    存储过程是一个命名程序块，包括过程的名称，过程使用的参数以及过程执行的操作。
    语法格式：创建存储过程
      CREATE [OR REPLACE] PROCEDURE
      过程名(参数1 [IN/OUT/IN OUT] 数据类型,参数2 [IN/OUT/IN OUT] 数据类型...)
      IS/AS
      PL/SQL过程体;
      (1)OR REPLACE:表示如果存储过程已经存在，则替换已有的存储过程。
      (2)IN 表示传入参数，不可以被赋值;OUT表示传出参数，可以被赋值;
        IN OUT表示传入传出参数，可以传入值，可以被赋值，可以返回值。
        如果这部分省略，默认表示传入参数，创建存储过程可以带参也可以不带参。
      (3)IS/AS:在IS/AS后声明变量不要加DECLARE语句。
      (4)参数的数据类型也是不能带长度。
    命名规范：
      存储过程命名规范:SP_目标表名
      存储过程传入参数命名规范:P_参数名(P_START_DATE)
      存储过程变量命名规范:V_变量名(V_END_DATE)

--语法结构 PROCEDURE 
CREATE OR REPLACE PROCEDURE SP_A IS
  --IS / AS
  --不能写DECLARE IS - BEGIN之间是声明部分
  V_A NUMBER;
BEGIN
  DBMS_OUTPUT.put_line('HELLO 张三');
END;

--调用

--CALL 存储过程名()小括号必须有;
CALL SP_A();

--代码块调用
BEGIN
--直接写存储过程名调用
  SP_A;
END;

--在命令行调用
EXECUTE 存储过程名;

--删除存储过程
DROP PROCEDURE SP_A;

------存储过程的参数 IN 入参 OUT 出参 IN OUT 出入参数

---入参 IN 默认入参
--传入一个员工编号,输出他的员工姓名
--P_参数名  参数类型  数据类型
CREATE OR REPLACE PROCEDURE SP_A(P_EMPNO IN NUMBER) IS
  V_ENAME VARCHAR2(20);
BEGIN
  SELECT ENAME INTO V_ENAME FROM EMP WHERE EMPNO = P_EMPNO;
  DBMS_OUTPUT.put_line(V_ENAME);
END;

--调用
BEGIN
  SP_A(7654);
END;

--出参 OUT
--存储过程向用户返回的结果
--传入一个员工编号,返回他的员工姓名
CREATE OR REPLACE PROCEDURE SP_A(P_EMPNO NUMBER, P_ENAME OUT VARCHAR2) IS
BEGIN
  --出参可以作为赋值目标使用,相当于变量,入参不可以
  SELECT ENAME INTO P_ENAME FROM EMP WHERE EMPNO = P_EMPNO;
END;

--调用
DECLARE
  V_ENAME VARCHAR2(20);
BEGIN
  --出参需要有赋值目标接收 先入参后出参
  --SP_A(7839,V_ENAME);
  SP_A(P_EMPNO => 7788, P_ENAME => V_ENAME); --不需要考虑顺序
  DBMS_OUTPUT.put_line(V_ENAME);
END;


--出入参数 IN OUT
--可以当入参使用,可以当出参使用
--传入一个员工编号,返回他的员工工资

BEGIN
  SELECT SAL
    INTO P_A /*出参*/

    小练习：
      创建一个存储过程，输入参数为员工编号，
      输出参数为输入员工所在部门名称和该部门平均工资，
      要求在处理过程中使用EMP1表作为员工表：
      1.若输入员工在ACCOUNTING部门，该员工职位为MANAGER，
      则该部门全员涨薪1000，该员工为其他职位，则该部门全员涨薪800；
      2.若输入员工在SALES部门，该员工职位为MANAGER，则该部门全员涨薪800，
      该员工为其他职位，则该部门全员涨薪600；
      3.若输入员工在其他部门，不管该员工什么职位，所在部门全员涨薪500；
      4.将上述涨薪操作更新EMP1表；
      5.平均工资为涨薪后平均工资。

CREATE OR REPLACE PROCEDURE SP_A(P_EMPNO IN NUMBER,
                                 P_DNAME OUT VARCHAR2,
                                 P_AVG   OUT NUMBER) IS
  V_JOB VARCHAR2(20);
  V_BM  NUMBER;
BEGIN
  SELECT DNAME, JOB, E.DEPTNO
    INTO P_DNAME, V_JOB, V_BM
    FROM EMP E
    JOIN DEPT D
      ON E.DEPTNO = D.DEPTNO
   WHERE EMPNO = P_EMPNO;
  IF P_DNAME = 'ACCOUNTING' THEN
    IF V_JOB = 'MANAGER' THEN
      UPDATE EMP1 SET SAL = SAL + 1000 WHERE DEPTNO = V_BM;
    ELSE
      UPDATE EMP1 SET SAL = SAL + 800 WHERE DEPTNO = V_BM;
    END IF;
  ELSIF P_DNAME = 'SALES' THEN
    IF V_JOB = 'MANAGER' THEN
      UPDATE EMP1 SET SAL = SAL + 800 WHERE DEPTNO = V_BM;
    ELSE
      UPDATE EMP1 SET SAL = SAL + 600 WHERE DEPTNO = V_BM;
    END IF;
  ELSE
    UPDATE EMP1 SET SAL = SAL + 500 WHERE DEPTNO = V_BM;
  END IF;
  SELECT AVG(SAL) INTO P_AVG FROM EMP1 WHERE DEPTNO = V_BM;
END;

--调用
DECLARE
  V_DNAME VARCHAR2(20);
  V_AVG   NUMBER;
BEGIN
  --7782 ACCOUNTING MANAGER + 1000
  SP_A(P_EMPNO => 7782, P_DNAME => V_DNAME, P_AVG => V_AVG);
  DBMS_OUTPUT.put_line(V_DNAME || V_AVG);
END;

--存在就更新,不存在就新增

1   张三  22

--MERGE INTO
/*
MERGE INTO 目标表名
USING      (SELECT 子查询) 源表数据
ON          (匹配条件)
WHTN MATCHED THEN
--存在就更新
UPDATE 不写表名 SET 字段名 = 值 WHERE可写可不写
WHEN NOT MATCHED THEN
--不存在就新增
INSERT (字段名.....) VALUES(值.....);
*/

CREATE OR REPLACE PROCEDURE SP_A(P_BM NUMBER) IS
BEGIN
  MERGE INTO EMP_MB M
  USING (SELECT * FROM EMP_YB Y WHERE Y.DEPTNO = P_BM) Y
  ON (M.EMPNO = Y.EMPNO)
  WHEN MATCHED THEN
    UPDATE
       SET --M.EMPNO = Y.EMPNO,比较字段不能更新
             M.ENAME = Y.ENAME,
           M.JOB      = Y.JOB,
           M.MGR      = Y.MGR,
           M.HIREDATE = Y.HIREDATE,
           M.SAL      = Y.SAL,
           M.COMM     = Y.COMM,
           M.DEPTNO   = Y.DEPTNO
  WHEN NOT MATCHED THEN
    INSERT
      (M.EMPNO, M.ENAME, M.JOB, M.MGR, M.HIREDATE, M.SAL, M.COMM, M.DEPTNO)
    VALUES
      (Y.EMPNO, Y.ENAME, Y.JOB, Y.MGR, Y.HIREDATE, Y.SAL, Y.COMM, Y.DEPTNO);

END;


--调用
BEGIN
  SP_A(99);
END;

CREATE OR REPLACE PROCEDURE SP_A(P_BM NUMBER) IS
  CURSOR C_A IS
    SELECT * FROM EMP_YB WHERE DEPTNO = P_BM;
  V_C NUMBER;
BEGIN
  FOR I IN C_A LOOP
    SELECT COUNT(1) INTO V_C FROM EMP_MB WHERE EMPNO = I.EMPNO;
    IF V_C = 0 THEN
      --不存在就新增
      INSERT INTO EMP_MB
      VALUES
        (I.EMPNO,
         I.ENAME,
         I.JOB,
         I.MGR,
         I.HIREDATE,
         I.SAL,
         I.COMM,
         I.DEPTNO);
    ELSE
      --存在就更新
      UPDATE EMP_MB M
         SET M.EMPNO    = I.EMPNO,
             M.ENAME    = I.ENAME,
             M.JOB      = I.JOB,
             M.MGR      = I.MGR,
             M.HIREDATE = I.HIREDATE,
             M.SAL      = I.SAL,
             M.COMM     = I.COMM,
             M.DEPTNO   = I.DEPTNO
       WHERE EMPNO = I.EMPNO;
    END IF;
  END LOOP;
END;

--调用
BEGIN
  SP_A(99);
END;
