表分区
    表分区的具体作用：
      分区功能可以将表、索引或者索引组织表进一步细分为段，这些数据对象的段叫作分区，
      每个分区都有自己的名称，一个分区后的对象(表)具体有多个段，这些段既可以集体管理，也可以单独管理。
    什么时候需要用分区表：
      (1)表的数据量特别大 2g
      (2)表中包含历史数据，新的数据被增加到新的分区中。
    表分区的优缺点：
      优点：
        1.改善查询性能：对分区对象的查询可以仅搜索自己关心的分区，提高检索速度。
        2.增强可用性：如果表的某个分区出现故障，表在其他分区的数据仍然可用。
        3.维护方便：如果表的某个分区出现故障，需要修复数据，只修复该分区即可。
        4.均衡I/O：可以把不同的分区映射到磁盘以平衡I/O，改善整个系统的性能。
      缺点：
********已经存在的表没有方法可以直接转化为分区表。
        2.需要维护。
    表分区的几种类型：
      1.范围分区：RANGE 针对数值、金额、年龄、时间等进行数据内容的划分，对相同范围的数据进行统一的管理
      2.列表分区：LIST 有大量的重复信息的列上面，例如省份、商品分类、性别、部门、岗位...
      3.散列分区  如果表格的字段没有任何的规则，那么就创建散列分区，数据库内部通过数据的hash哈希算法，将数据计算能不同的哈希值，然后根据计算的结果进行数据的分类。
      4.组合分区：范围分区+列表分区 列表分区+范围分区 

-----语法结构 PARTITION BY 
--范围
CREATE TABLE EMP_FQ(
EMPNO NUMBER,
ENAME VARCHAR2(20),
SAL   NUMBER,
DEPTNO NUMBER
) PARTITION BY RANGE(SAL)(
PARTITION F1 VALUES LESS THAN(1000),--小于
--下一个分区的界限值一定要大于上一个界限值
PARTITION F2 VALUES LESS THAN(2000),--1000-1999
PARTITION F3 VALUES LESS THAN(MAXVALUE) --MAXVALUE最大值
);

--新增数据一定要符合分区已设置的分区条件
INSERT INTO EMP_FQ SELECT EMPNO,ENAME,SAL,DEPTNO FROM EMP;

--查寻分区表
SELECT * FROM EMP_FQ PARTITION(F2) WHERE DEPTNO = 20;

--列表分区 LIST
CREATE TABLE EMP_FQ
PARTITION BY LIST(DEPTNO)(
PARTITION F1 VALUES(10),
PARTITION F2 VALUES(20),
PARTITION F3 VALUES(30,40),
PARTITION F4 VALUES(DEFAULT)
)
AS SELECT * FROM EMP;

--查寻分区
SELECT * FROM EMP_FQ PARTITION(F3);

--散列 HASH
CREATE TABLE EMP_FQ
PARTITION BY HASH(ENAME) PARTITIONS 3
AS SELECT * FROM EMP;

--2.0
CREATE TABLE EMP_FQ
PARTITION BY HASH(ENAME)(
PARTITION F1 ,
PARTITION F2 ,
PARTITION F3
)
AS SELECT * FROM EMP;

--查寻分区表
SELECT *  FROM EMP_FQ PARTITION(F1);

--组合分区
范围(主)+列表(子)

CREATE TABLE EMP_FQ
--先声明主分区类型
PARTITION BY RANGE(SAL)
--再声明子分区类型 SUBPARTITION
SUBPARTITION BY LIST(DEPTNO)(
--先设置主分区条件
PARTITION F1 VALUES LESS THAN(1000)(
--再设置子分区条件
SUBPARTITION Z1 VALUES(10),
SUBPARTITION Z2 VALUES(20),
SUBPARTITION Z3 VALUES(30)
),
PARTITION F2 VALUES LESS THAN(3000)(
SUBPARTITION Z4 VALUES(10),
SUBPARTITION Z5 VALUES(20,30)
),
PARTITION F3 VALUES LESS THAN(MAXVALUE)(
SUBPARTITION Z6 VALUES(10),
SUBPARTITION Z7 VALUES(20),
SUBPARTITION Z8 VALUES(30)
))
AS SELECT * FROM EMP;

--查寻主分区
SELECT * FROM EMP_FQ PARTITION(F2);

--查寻子分区
SELECT * FROM EMP_FQ SUBPARTITION(Z5);



--范围
CREATE TABLE EMP_FQ1 PARTITION BY RANGE(SAL)(
PARTITION F1 VALUES LESS THAN(1000),
PARTITION F2 VALUES LESS THAN(2000),
PARTITION F3 VALUES LESS THAN(MAXVALUE)
) AS SELECT * FROM EMP;
--列表分区 LIST
CREATE TABLE EMP_FQ2
PARTITION BY LIST(DEPTNO)(
PARTITION F1 VALUES(10),
PARTITION F2 VALUES(20),
PARTITION F3 VALUES(30,40)
)
AS SELECT * FROM EMP;
--散列
CREATE TABLE EMP_FQ3
PARTITION BY HASH(ENAME)(
PARTITION F1 ,
PARTITION F2 ,
PARTITION F3
)
AS SELECT * FROM EMP;
---组合分区
CREATE TABLE EMP_FQ4
PARTITION BY RANGE(SAL)
SUBPARTITION BY LIST(DEPTNO)(
PARTITION F1 VALUES LESS THAN(1000)(
SUBPARTITION Z1 VALUES(10),
SUBPARTITION Z2 VALUES(20),
SUBPARTITION Z3 VALUES(30)
),
PARTITION F2 VALUES LESS THAN(3000)(
SUBPARTITION Z4 VALUES(10),
SUBPARTITION Z5 VALUES(20,30)
),
PARTITION F3 VALUES LESS THAN(MAXVALUE)(
SUBPARTITION Z6 VALUES(10),
SUBPARTITION Z7 VALUES(20),
SUBPARTITION Z8 VALUES(30)
))
AS SELECT * FROM EMP;

---新增分区

--范围
--ALTER TABLE 表名 ADD PARTITION 分区名 VALUES LESS THAN(界限值);
--新分区的界限值要大于上一个分区
ALTER TABLE EMP_FQ1 ADD PARTITION F4 VALUES LESS  THAN(8000);

--列表
--ALTER TABLE 表名 ADD PARTITION 分区名 VALUES(界限值);
ALTER TABLE EMP_FQ2 ADD PARTITION F4 VALUES(80);

--散列 新增分区后数据会重新分配
--ALTER TABLE 表名 ADD PARTITION 分区名;
ALTER TABLE EMP_FQ3 ADD PARTITION F4;

--组合分区
--主分区  只有一个默认的子分区
ALTER TABLE EMP_FQ4 ADD PARTITION F4 VALUES LESS  THAN(8000);

--子分区没有新增操作


---删除分区 删除分区结构,删除分区数据
--ALTER TABLE 表名 DROP (SUB)PARTITION 分区名;
ALTER TABLE EMP_FQ4 DROP SUBPARTITION Z1;
--散列分区不能删除,拆分,合并


---重命名分区
--ALTER TABLE 表名 RENAME (SUB)PARTITION 旧 TO 新;
ALTER TABLE EMP_FQ4 RENAME PARTITION F1 TO AA;


---截断分区 清空分区数据,保留分区结构
--ALTER TABLE 表名 TRUNCATE PARTITION 分区名;
ALTER TABLE EMP_FQ1 TRUNCATE (SUB)PARTITION AA;



--拆分分区 SPLIT
--ALTER TABLE 表名 SPLIT PARTITION 分区名 AT/VALUES(界限值)
--INTO (PARTITION 分区1 , PARTITION 分区2);

--范围
ALTER TABLE EMP_FQ1 SPLIT PARTITION F3 AT (8000) INTO
(PARTITION AA,PARTITION BB);

--列表 10 20 30
ALTER TABLE EMP_FQ2 SPLIT PARTITION F3 VALUES(30) INTO
(PARTITION AA,PARTITION BB);

--组合分区
--主   子分区数量是复制的
ALTER TABLE EMP_FQ4 SPLIT PARTITION F3 AT (8000) INTO
(PARTITION AA,PARTITION BB);

--子分区
ALTER TABLE EMP_FQ4 SPLIT SUBPARTITION Z5 VALUES(30) INTO
(SUBPARTITION CC,SUBPARTITION DD);


---合并分区 MERGE INTO
--ALTER TABLE 表名 MERGE PARTITIONS 分区1,分区2 INTO PARTITION 分区;

--范围
--界限值大的分区名可以作为结果分区名重复只用
--只能合并两个相邻的分区
ALTER TABLE EMP_FQ1 MERGE PARTITIONS WW,BB INTO  PARTITION XX;

--列表
ALTER TABLE EMP_FQ2 MERGE PARTITIONS F1,BB INTO  PARTITION WW;

--组合分区
--主  合并之后子分区只有一个默认分区
ALTER TABLE EMP_FQ4 MERGE PARTITIONS AA,BB INTO  PARTITION XX;

--子分区
--不能跨主分区合并子分区
ALTER TABLE EMP_FQ4 MERGE SUBPARTITIONS PP,CC INTO  SUBPARTITION M;


---WITH AS 短语 创建临时表 
查询出来的结果集会被放入到临时表中，注意临时表的空间，
数据量太大（几千万的数据量）可能会报内存不足的错误
WITH 可以将复杂的逻辑先写到 WITH AS 的子查询里。

--语法结构
WITH AA AS
 (SELECT * FROM EMP)
SELECT * FROM AA;


--查询工资最高的员工信息
--1.0
SELECT *
  FROM (SELECT E.*, ROW_NUMBER() OVER(ORDER BY SAL DESC) PM FROM EMP E)
 WHERE PM = 1;

--2.0
WITH AA AS
 (SELECT E.*, ROW_NUMBER() OVER(ORDER BY SAL DESC) PM FROM EMP E)
SELECT * FROM AA WHERE PM = 1;


--定义多个临时表
--1.0
SELECT *
  FROM (SELECT * FROM EMP) A
  LEFT JOIN (SELECT * FROM DEPT) B
    ON A.DEPTNO = B.DEPTNO;

--2.0
WITH AA AS
 (SELECT * FROM EMP),
BB AS
 (SELECT * FROM DEPT)
SELECT * FROM AA JOIN BB ON AA.DEPTNO = BB.DEPTNO;
