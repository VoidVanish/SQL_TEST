--使用FRO循环+PUT输出语句打印正三角

    *       1*2-1
   ***      2*2-1
  *****     3*2-1
 ******* 
********* 
 
DECLARE
  V_A NUMBER := &输入行号;
BEGIN
  --A=12345
  FOR A IN 1 .. V_A LOOP
    --------空格
    FOR B IN 1 .. V_A - A LOOP
      DBMS_OUTPUT.put(' ');
    END LOOP;
    ---------星号
    FOR I IN 1 .. 2 * A - 1 LOOP
      DBMS_OUTPUT.put('*');
    END LOOP;
    DBMS_OUTPUT.new_line;
  END LOOP;
END;

----游标
    游标是指向查询结果集的一个指针，通过游标可以将查询结果集中的记录逐一取出，
    并在PL/SQL程序块中进行处理。
    游标的类型有两种：隐式游标和显式游标。隐式游标是由系统自动创建并管理的游标。
    PL/SQL会为所有的sql数据操作声明一个隐式的游标，包括只返回一条记录的查询操作。
    对于返回多条记录的查询，必须自己创建显式游标。


--显式游标 CURSOR
DECLARE
  --CURSOR C_游标名 IS SELECT 查询语句;
  CURSOR C_A IS
    SELECT * FROM EMP;
  V_A EMP%ROWTYPE;
BEGIN
  OPEN C_A; --打开游标
  --只能将游标中的数据逐行赋值给变量,再通过变量使用游标内数据
  --FETCH 游标名 INTO 变量名; INTO两边字段数量数据类型保持一致
  LOOP
    FETCH C_A
      INTO V_A;
    EXIT WHEN C_A%NOTFOUND; --退出循环条件
    DBMS_OUTPUT.put_line(V_A.ENAME || V_A.SAL);
  END LOOP;
  --游标不能直接输出  
  --DBMS_OUTPUT.put_line(C_A);
  CLOSE C_A; --关闭游标
END;

--游标属性：*********游标名%属性名
%FOUND 布尔型属性，当最近一次读取记录时成功返回，则值为TRUE，否则为FALSE。
%NOTFOUND  布尔型属性，与%FOUND相反。
%ISOPEN  布尔型属性，当游标已打开时返回TRUE，否则返回FALSE。
%ROWCOUNT  数字型属性，返回已从游标中读取的记录数。

--游标搭配WHILE循环使用

DECLARE
  CURSOR C_A IS
    SELECT * FROM EMP;
  V_A EMP%ROWTYPE;
BEGIN
  OPEN C_A;
  FETCH C_A
    INTO V_A; --进入循环的条件
  --只有进行了查找操作,才能判断是都找到数据 
  WHILE C_A%FOUND LOOP
    DBMS_OUTPUT.put_line(V_A.ENAME || V_A.SAL);
    FETCH C_A
      INTO V_A; --获取游标内下一条数据
  END LOOP;
  CLOSE C_A;
END;

--带参游标 1.定义参数 2.传参
--一个参数就是一个过滤条件

传入一个部门编号,输出该部门的姓名和工资

DECLARE
  --P_参数名 数据类型(不能加长度)
  CURSOR C_A(P_BM NUMBER,P_SAL NUMBER) IS
    SELECT * FROM EMP WHERE DEPTNO = P_BM
    AND SAL > P_SAL;
  V_A EMP%ROWTYPE;
BEGIN
--参数定义后一定要传参
  OPEN C_A(P_BM => 20,P_SAL => 2000);
  LOOP
    FETCH C_A
      INTO V_A;
    EXIT WHEN C_A%NOTFOUND;
    DBMS_OUTPUT.put_line(V_A.ENAME || V_A.DEPTNO);
  END LOOP;
  CLOSE C_A;
END;


小练习：
对于EMP1表中SALESMAN和MANAGER，
如果这个员工工资小于1000，就给他加300奖金
如果这个员工工资1000到2000，就给他加200奖金
如果这个员工工资大于2000，就给他加100奖金
输出员工编号，员工名字，员工工资，员工奖金
        
DECLARE
  CURSOR C_A IS
    SELECT * FROM EMP WHERE JOB IN ('SALESMAN', 'MANAGER');
  V_A EMP%ROWTYPE;
BEGIN
  OPEN C_A;
  LOOP
    FETCH C_A
      INTO V_A;
    EXIT WHEN C_A%NOTFOUND;
    IF V_A.SAL < 1000 THEN
      V_A.COMM := NVL(V_A.COMM, 0) + 300;
    ELSIF V_A.SAL <= 2000 THEN
      V_A.COMM := NVL(V_A.COMM, 0) + 200;
    ELSE
      V_A.COMM := NVL(V_A.COMM, 0) + 100;
    END IF;
    DBMS_OUTPUT.put_line(V_A.EMPNO || V_A.ENAME || V_A.SAL || '  ' ||
                         V_A.COMM);
  END LOOP;
  CLOSE C_A;
END;
        
--游标搭配FOR循环使用
PL/SQL语言提供了游标FOR循环语句，自动执行游标OPEN、FETCH、CLOSE语句和
循环语句的功能；当进入循环时，游标FOR循环语句自动打开游标，并提取第一行游标数据，
当程序处理完当前所提取的数据而进入下一次循环时，游标FOR循环语句自动提取下一行数据
供程序处理，当提取完结果集中的所有数据行后结束循环，并自动关闭游标。

DECLARE
  CURSOR C_A(P_BM NUMBER) IS
    SELECT * FROM EMP WHERE DEPTNO = P_BM;
BEGIN
  --FOR I IN 游标名 LOOP
  FOR I IN C_A(20) LOOP
    DBMS_OUTPUT.put_line(I.ENAME || I.SAL || I.JOB);
  END LOOP;
END;

--FOR循环支持使用子查询代替游标
BEGIN
  FOR I IN (SELECT * FROM EMP) LOOP
    DBMS_OUTPUT.put_line(I.ENAME || I.SAL || I.JOB);
  END LOOP;
END;

DECLARE

END;

--隐式游标
显式游标主要是用于对查询语句的处理，尤其是在查询结果为多条记录的情况下；
而对非查询语句如修改、删除(DML)操作，则由oracle自动为这些操作设置游标，
这些由系统隐含创建的游标称为隐式游标，隐士游标的名字为SQL，这些由oracle系统定义的隐式游标的操作，
如定义、打开、取值及关闭操作，都由oracle自动完成，无需用户处理。
用户只能通过隐式游标的相关属性，来完成相应的操作。

SQL%ISOPEN 布尔型属性，取值总是FALSE，SQL命令执行完毕立即关闭隐式游标。
SQL%FOUND 布尔型属性，当最近一次读取记录时成功返回，则值为TRUE，否则为FALSE。
SQL%NOTFOUND  布尔型属性，与SQL%FOUND相反。
SQL%ROWCOUNT  数字型属性，返回已从游标中读取的记录数。

--动态SQL
    在PL/SQL程序开发中，可以使用DML语句，但是很多语句(如DDL)
    不能直接在PL/SQL中执行，这些语句可以使用动态SQL来实现。
    使用动态方式，可以执行任何类型的sql语句。
    语法格式：
      EXECUTE IMMEDIATE 动态语句字符串
      [INTO 变量列表]可选 SELECT 查询语句
      [USING 参数列表] 可选


--语法结构
--使用动态sql运行DDL语句
BEGIN
  EXECUTE IMMEDIATE 'DROP TABLE EMP1';
  --DROP TABLE EMP1
END;

--使用动态SQL执行DML语句
BEGIN
 EXECUTE IMMEDIATE ' DELETE FROM EMP1';
END;

--传入一个员工编号,删除该员工信息
DECLARE
  V_EMPNO NUMBER := &编号;
BEGIN
  EXECUTE IMMEDIATE 'DELETE FROM EMP1 WHERE  EMPNO = ' || V_EMPNO;
  --输出语句输出什么内容,动态SQL就执行什么内容
  DBMS_OUTPUT.put_line('DELETE FROM EMP1 WHERE  EMPNO = ' || V_EMPNO);
END;

--传入一个员工姓名,删除该员工信息
DECLARE
  V_ENAME VARCHAR2(20) := '&姓名';
BEGIN
  EXECUTE IMMEDIATE 'DELETE FROM EMP1 WHERE ENAME = ' ||
                       CHR(39) || V_ENAME ||CHR(39);
  DBMS_OUTPUT.put_line('DELETE FROM EMP1 WHERE ENAME = ' ||
                       '''' || V_ENAME ||'''');
END;
