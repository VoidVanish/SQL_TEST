--返回毫秒

--获取日期中的月month('日期')

select month('2026-06-07');

--获取日期的天 day
select day('2026-06-07');

--获取小时hour
select hour('2026-06-07 17:30:23');

--获取两个日期相差的天数datediff(日期A,日期B)

select datediff('2026-06-06','2025-06-06');

--日期加天数 date_add(日期,数字)

select date_add('2026-06-06',6);

--日期减天数 date_sub(日期,数字)

select date_sub('2026-06-06',6);

--指定日期格式 date_format(日期,格式)

select date_format('2026-11-07','yyyy年-MM月-dd日');

集合函数
map 创建map集合  参数数量是偶数
select map('语文','78','数学','87');

map_keys获取map集合中的key值
select MAP_KEYs(map('语文','78','数学','87'));

map_values获取map集合中的value值
select MAP_values(map('语文','78','数学','87'));

--array 生成一个数组
select array('张三','李四','王五');

--sort_array 对数组内的值进行排序
select sort_array(array('c','d','a'));

--struct 生成一个struct集合数据类型
select struct('18','19','20');

--named_struct 可以声明struct类型的key值以及value值

select named_struct('语文','78','数学','87');
