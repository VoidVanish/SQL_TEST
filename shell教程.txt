3.变量
	3.1自定义变量
		变量名和等号之间不能有空格
		命名规则：只能使用英文字母，数字，下划线，且首个字符不能以数字开头
		使用变量：$变量名，${变量名}
		3.1.2只读变量
			readonly 变量名
			只读变量的值不能被改变
		3.1.3删除变量
			unset 变量名
			变量被删除后就不能被使用
			unset不能删除只读变量
	3.2系统变量
		$HOME 家目录
		$PWD 当前执行位置
		$USER 用户
		$SHELL 使用的解释器
	3.3特殊变量与传参
		$# 传递参数的个数
		$n 每个传递的参数：$0 脚本名称，$1第一个参数，$2第二个参数,${10} 第10个参数及以后的参数都需要加{}
		$$ 脚本运行的当前进程ID（PID）
		$? 显示最后命令的退出状态，0表示没有错误，其他任何值表示有错误
		$* 以一个单字符串显示所有向脚本传递的参数。如果用""括起来"$*"，以"$1 $2 $3...$n"的形式输出所有参数  
		$@ 与$*相同，但是使用时加引号，并在引号中返回每个参数。如"$@"用"括起来的情况，以"$1" "$2" "$3" ..."$n"的形式输出所有参数
4.数据类型
	4.1字符串
		"" 双引号，弱引用，引号内的特殊字符有意义
		'' 单引号，强引用，引号内的特殊字符取消意义
		4.1.2字符串的拼接
			双引号拼接
			your_name=" zhangsan"
			sp1="hello,"$your_name"!"
			sp2="hello,$your_name!"
			echo $sp1 $sp2
			单引号拼接
			dp1='hello,'$your_name'!'
			dp2='hello,$your_name!'
			echo $dp1 $dp2
		4.1.3字符串的截取
			获取变量字符串长度
			${#字符串}
			实例：${#strc}
			切割字符串
			${字符串:(大于)起始位置:切割个数}
			实例：echo ${strc:1:4}
			查找字符i或者o的位置（哪个先出现就计算哪个）
			expr index "$strc" io
	4.2 数组
		定义数组
			array=(value0 value1 value2 value3)
		单独定义数组
			array[0]=value0
			array[1]=value3
			array[15]=value15
		读取数组
			valuen=${array_name[n]}
		获取数组长度
			获取数组元素个数
				len_a=${#array_name[@]} 或者 len_a=${#array_name[*]}
			获取数组单个元素长度
				lenn=${#array_name[n]}
5.运算符
	5.1算术运算符：$((运算式)),$[运算式],`expr 运算式` 运算时空格隔开
		+ 加法 `expr $a + $b`
		- 减法 `expr $a - $b`
		* 乘法 `expr $a \* $b`
		/ 除法 `expr $a / $b`
		% 取余 `expr $a % $b`
		== 相等。比较两个数字，相同返回TRUE，不相同返回FALSE。 [ $a == $b ]
		!= 不相等。比较两个数字，不相同返回TRUE，相同返回FALSE。 [ $a != $b ]
	5.2关系运算符
		-lt 小于(less than) [ $a -lt $b ]   
		-le 小于等于(less equal) [ $a -le $b ]
		-gt 大于(greater than) [ $a -gt $b ]
		-ge 大于等于(greater equal) [ $a -ge $b ]
		-eq 等于(equal) [ $a -eq $b ]
		-ne 不等于(not equal) [ $a -ne $b ]
	5.3逻辑运算符
		-a 与运算（且） [ $a -lt 10 -a $b -gt $a ]
		-o 或运算（或） [ $a -lt 10 -o $b -lt 10 ]
		! 非运算（非） [ ! $a -ge $b ]
	5.4字符串运算
		= 检测两个字符串是否相等，相等返回true。 [ $a = $b ]
		!= 检测两个字符串是否不相等，不相等返回true。 [ $a != $b ]
		-z 检测字符串长度是否为0，为0返回true。 [ -z $a ]
		-n 检测字符串长度是否不为0，不为0返回true。 [ -n "$a" ]
		$ 检测字符串是否不为空，不为空返回true。 [ $a ]
	5.5文件测试运算符
		file="/home/hadoop/123/test.sh"
		-d 检查文件是否是目录，如果是，返回true。 [ -d $file ]
		-f 检查文件是否为普通文件，如果是，返回true。 [ -f $file ]
		-s 检查文件是否为空，如果不为空，返回true。 [ -s $file ]
		-e 检查文件（包括目录）是否存在，如果是，返回true。 [ -e $file]
		-r 检查文件是否可读，如果是，返回true。 [ -r $file ]
		-w 检查文件是否可写，如果是，返回true。 [ -r $file ]
		-x 检查文件是否可执行，如果是，返回true。 [-x $file ]

	6.3for循环
		6.3.1((首位;末位;步长))
			s=0
			for ((i=0;i<=100;i+=1))
			do
				s=$[ $s + $i ]
				或
				s=`expr $s + $i`
			done
		6.3.2$(seq 首位 步长 末位)
			s=0
			for i in $(seq 0 1 100)
			do
				s=$[ $s + $i ]
				或
				s=`expr $s + $i`
			done
		6.3.3in "$*"与"$@"区别
			for i in "$*"
			do
				echo $i
			done
			
			for i in "$@"
			do
				echo $i
			done
	6.4while循环
		s=0
		i=0
		while [ $i -le 100 ]
		do
			echo
		done
	6.5跳出循环
		6.5.1continue跳出单次循环
			s=0
			i=0
			while [ $i -lt 100 ]
			do
					i=$[ $i + 1 ]
					yu=`expr $i % 2`
					if [ $yu -gt 0 ]
					then
							continue
					fi
					s=$[ $s + $i ]
			done
			echo "1到100偶数相加等于$s"
		6.5.2break跳出全部循环
			s=0
			i=0
			while [ $i -lt 100 ]
			do
					i=$[ $i + 1 ]
					if [ $i -gt 10 ]
					then
							break
					fi
					s=$[ $s + $i ]
			done
			echo "1到10相加等于$s"
